Q2. Write a parameterised cursor that accepts a publisher name and lists the title and price of every book published by that publisher. Make the comparison case-insensitive using UPPER().

//answer

SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book(p_pub_name VARCHAR2) IS
        SELECT b.title, b.price
        FROM book b
        JOIN publisher p
        ON b.pub_id = p.pub_id
        WHERE UPPER(p.pub_name) = UPPER(p_pub_name);

BEGIN
    FOR r IN c_book('Tech Bharat Pub') LOOP
        DBMS_OUTPUT.PUT_LINE(
            r.title || ' - Rs.' || r.price
        );
    END LOOP;
END;
