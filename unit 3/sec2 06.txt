Q6. Fetch all books and calculate the total stock value (price * stock) by accumulating inside the loop. Display each book's individual value in the loop and the grand total after the loop ends.

//answer

SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book IS
        SELECT book_id, title, price, stock
        FROM book;

    v_value NUMBER;
    v_total NUMBER := 0;

BEGIN
    FOR r IN c_book LOOP
        v_value := r.price * r.stock;
        v_total := v_total + v_value;

        DBMS_OUTPUT.PUT_LINE(
            r.book_id || ' - ' ||
            r.title || ' - Value = Rs.' ||
            v_value
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE(
        'Grand Total = Rs.' || v_total
    );
END;
