Q9. Write a parameterised cursor that accepts a month number (1-12) and lists all books issued in that month of 2026. Use EXTRACT(MONTH FROM issue_date).
//answer

SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_issue(p_month NUMBER) IS
        SELECT b.title, bi.issue_date
        FROM book_issue bi
        JOIN book b
        ON bi.book_id = b.book_id
        WHERE EXTRACT(MONTH FROM bi.issue_date) = p_month
        AND EXTRACT(YEAR FROM bi.issue_date) = 2026;

BEGIN
    FOR r IN c_issue(6) LOOP
        DBMS_OUTPUT.PUT_LINE(
            r.title || ' - ' ||
            TO_CHAR(r.issue_date, 'DD-MON-YYYY')
        );
    END LOOP;
END;
/
