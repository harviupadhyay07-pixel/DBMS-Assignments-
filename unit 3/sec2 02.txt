Q2. Rewrite the same program using a cursor FOR loop. In one line of comment, state which steps Oracle performed for you automatically.

//answer

SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book IS
        SELECT book_id, title, price
        FROM book;

BEGIN
    FOR r IN c_book LOOP
        DBMS_OUTPUT.PUT_LINE(
            r.book_id || ' - ' || r.title || ' - Rs.' || r.price
        );
    END LOOP;

    -- Oracle automatically performs OPEN, FETCH and CLOSE.
END;
