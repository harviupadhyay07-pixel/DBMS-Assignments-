New Tax Regime 2024-25: Deduct standard Rs.75,000 first. Then calculate tax using the given slabs: 0-3L=NIL, 3-7L=5%, 7-10L=10%, 10-12L=15%, 12-15L=20%, >15L=30%. Calculate gross salary, taxable income, total tax, monthly TDS and take-home monthly salary. Use SEARCHED CASE statement.

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_gross_salary NUMBER := 1000000;
    v_taxable_income NUMBER;
    v_tax NUMBER := 0;
    v_monthly_tds NUMBER;
    v_take_home NUMBER;
BEGIN
    v_taxable_income := v_gross_salary - 75000;

    CASE
        WHEN v_taxable_income <= 300000 THEN
            v_tax := 0;

        WHEN v_taxable_income <= 700000 THEN
            v_tax := (v_taxable_income - 300000) * 0.05;

        WHEN v_taxable_income <= 1000000 THEN
            v_tax := 20000 +
                     (v_taxable_income - 700000) * 0.10;

        WHEN v_taxable_income <= 1200000 THEN
            v_tax := 50000 +
                     (v_taxable_income - 1000000) * 0.15;

        WHEN v_taxable_income <= 1500000 THEN
            v_tax := 80000 +
                     (v_taxable_income - 1200000) * 0.20;

        ELSE
            v_tax := 140000 +
                     (v_taxable_income - 1500000) * 0.30;
    END CASE;

    v_monthly_tds := v_tax / 12;
    v_take_home := (v_gross_salary - v_tax) / 12;

    DBMS_OUTPUT.PUT_LINE('Gross Salary: Rs.' || v_gross_salary);
    DBMS_OUTPUT.PUT_LINE('Taxable Income: Rs.' || v_taxable_income);
    DBMS_OUTPUT.PUT_LINE('Total Tax: Rs.' || v_tax);
    DBMS_OUTPUT.PUT_LINE('Monthly TDS: Rs.' || ROUND(v_monthly_tds));
    DBMS_OUTPUT.PUT_LINE('Take Home Monthly Salary: Rs.' || ROUND(v_take_home));

    IF v_tax = 0 THEN
        DBMS_OUTPUT.PUT_LINE('No tax this year — save more with PPF/ELSS!');
    END IF;
END;
