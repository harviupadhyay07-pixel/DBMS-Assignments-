p2.8

Build an HDFC/SBI style loan portal. Input monthly_salary, existing_emi, loan_amount, annual_rate and tenure_months. FOIR limit is 40% of salary. Minimum salary is Rs.25,000 and maximum loan is 60 times monthly salary. Calculate EMI using the given formula. If EMI is above FOIR, use WHILE LOOP to reduce the loan amount by Rs.10,000 until eligible. Print calculated EMI, FOIR limit, approved amount and decision.

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_monthly_salary NUMBER := 50000;
    v_existing_emi NUMBER := 5000;
    v_loan_amount NUMBER := 1500000;
    v_annual_rate NUMBER := 10;
    v_tenure_months NUMBER := 60;

    v_monthly_rate NUMBER;
    v_emi NUMBER;
    v_foir_limit NUMBER;
    v_available_emi NUMBER;
    v_approved_amount NUMBER;
    v_decision VARCHAR2(20);
BEGIN
    v_monthly_rate := v_annual_rate / 1200;
    v_foir_limit := v_monthly_salary * 0.40;
    v_available_emi := v_foir_limit - v_existing_emi;

    v_monthly_rate := v_annual_rate / 1200;

    v_emi := v_loan_amount * v_monthly_rate
             * POWER(1 + v_monthly_rate, v_tenure_months)
             / (POWER(1 + v_monthly_rate, v_tenure_months) - 1);

    v_approved_amount := v_loan_amount;

    IF v_monthly_salary < 25000 THEN
        v_decision := 'REJECTED';
        v_approved_amount := 0;

    ELSIF v_emi <= v_available_emi THEN
        v_decision := 'APPROVED';

    ELSE
        WHILE v_emi > v_available_emi
              AND v_approved_amount > 0 LOOP

            v_approved_amount := v_approved_amount - 10000;

            v_emi := v_approved_amount * v_monthly_rate
                     * POWER(1 + v_monthly_rate, v_tenure_months)
                     / (POWER(1 + v_monthly_rate, v_tenure_months) - 1);
        END LOOP;

        v_decision := 'CONDITIONAL';
    END IF;

    DBMS_OUTPUT.PUT_LINE('Monthly Salary: Rs.' || v_monthly_salary);
    DBMS_OUTPUT.PUT_LINE('Existing EMI: Rs.' || v_existing_emi);
    DBMS_OUTPUT.PUT_LINE('FOIR Limit: Rs.' || v_foir_limit);
    DBMS_OUTPUT.PUT_LINE('Requested Loan: Rs.1500000');
    DBMS_OUTPUT.PUT_LINE('Approved Loan: Rs.' || v_approved_amount);
    DBMS_OUTPUT.PUT_LINE('Calculated EMI: Rs.' || ROUND(v_emi));
    DBMS_OUTPUT.PUT_LINE('Decision: ' || v_decision);
END;
