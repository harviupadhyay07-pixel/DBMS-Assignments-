p2.4

Part A: Print multiplication table of (roll_no MOD 9 + 2) from 1 to 20. Part B: Print tables of 2, 3, 4, 5 using nested FOR loops. Part C: Print right-angled star triangle with height = 5 + roll_no MOD 5. Part D: Print number pyramid for 6 rows. Part E: Print the same triangle in reverse using REVERSE keyword.

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_roll NUMBER := 56;
    v_table NUMBER := MOD(v_roll, 9) + 2;
    v_height NUMBER := 5 + MOD(v_roll, 5);
BEGIN
    DBMS_OUTPUT.PUT_LINE('Part A: Table of ' || v_table);

    FOR i IN 1..20 LOOP
        DBMS_OUTPUT.PUT_LINE(
            v_table || ' x ' || i || ' = ' || v_table * i
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Part B: Tables of 2, 3, 4 and 5');

    FOR i IN 2..5 LOOP
        FOR j IN 1..10 LOOP
            DBMS_OUTPUT.PUT_LINE(
                i || ' x ' || j || ' = ' || i * j
            );
        END LOOP;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Part C: Star Pattern');

    FOR i IN 1..v_height LOOP
        DBMS_OUTPUT.PUT_LINE(RPAD('*', i, '*'));
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Part D: Number Pyramid');

    FOR i IN 1..6 LOOP
        FOR j IN 1..i LOOP
            DBMS_OUTPUT.PUT(j);
        END LOOP;
        DBMS_OUTPUT.NEW_LINE;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Part E: Reverse Star Pattern');

    FOR i IN REVERSE 1..v_height LOOP
        DBMS_OUTPUT.PUT_LINE(RPAD('*', i, '*'));
    END LOOP;
END;
