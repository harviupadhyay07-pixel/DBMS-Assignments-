Problem Statement: Build an IRCTC-style ticket generator. Inputs: passenger name, dob, roll_no. Steps: (1) PNR = 'GJ' || TO_CHAR(SYSDATE,'YYMM') || LPAD(roll_no, 4, '0'), (2) journey date = SYSDATE + (roll_no MOD 30 + 1), printed as Day, DD-MON-YYYY, (3) base fare = roll_no * 50 + 200, (4) age = TRUNC(MONTHS_BETWEEN(SYSDATE, dob)/12); senior citizens (60+) get 40% concession — compute final fare with NVL2-style logic, (5) print the full ticket aligned with RPAD/LPAD like a printed counter ticket.

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_name VARCHAR2(50) := 'Harvi Upadhyay';
    v_dob DATE := DATE '2005-06-15';
    v_roll NUMBER := 56;
    v_pnr VARCHAR2(30);
    v_jdate DATE;
    v_base_fare NUMBER;
    v_age NUMBER;
    v_final_fare NUMBER;
BEGIN
    v_pnr := 'GJ' ||
             TO_CHAR(SYSDATE, 'YYMM') ||
             LPAD(v_roll, 4, '0');

    v_jdate := SYSDATE +
               (MOD(v_roll, 30) + 1);

    v_base_fare := v_roll * 50 + 200;

    v_age := TRUNC(
        MONTHS_BETWEEN(SYSDATE, v_dob) / 12
    );

    IF v_age >= 60 THEN
        v_final_fare := v_base_fare * 0.60;
    ELSE
        v_final_fare := v_base_fare;
    END IF;

    DBMS_OUTPUT.PUT_LINE('----- RAILWAY TICKET -----');
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Passenger:', 15) || v_name
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('PNR:', 15) || v_pnr
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Journey:', 15) ||
        TO_CHAR(
            v_jdate,
            'Day, DD-MON-YYYY'
        )
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Age:', 15) || v_age
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Base Fare:', 15) ||
        'Rs.' || v_base_fare
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Final Fare:', 15) ||
        'Rs.' || v_final_fare
    );
END;
