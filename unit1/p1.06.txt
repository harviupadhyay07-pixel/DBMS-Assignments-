Problem Statement: Build a bank interest calculator. Variables: principal, rate (%), years. Calculate: (1) Simple Interest = P*R*T/100, (2) Compound Interest = P*(1+R/100)^T - P using POWER function, (3) Final amounts for both. Print a comparison: show SI vs CI difference. If CI > SI by more than 10%: print "Tip: FD is better than savings account for this amount." Use TO_CHAR to format amounts with commas.

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_p NUMBER := 56000;
    v_r NUMBER := 7;
    v_t NUMBER := 3;
    v_si NUMBER;
    v_ci NUMBER;
    v_si_amount NUMBER;
    v_ci_amount NUMBER;
    v_diff_pct NUMBER;
BEGIN
    v_si := v_p * v_r * v_t / 100;

    v_ci := v_p *
        POWER(1 + v_r / 100, v_t) - v_p;

    v_si_amount := v_p + v_si;
    v_ci_amount := v_p + v_ci;

    v_diff_pct := (v_ci - v_si) / v_si * 100;

    DBMS_OUTPUT.PUT_LINE('Principal: ' ||
        TO_CHAR(v_p, '99,99,999'));
    DBMS_OUTPUT.PUT_LINE('Rate: ' || v_r || '%');
    DBMS_OUTPUT.PUT_LINE('Simple Interest: Rs.' ||
        ROUND(v_si, 2));
    DBMS_OUTPUT.PUT_LINE('Compound Interest: Rs.' ||
        ROUND(v_ci, 2));
    DBMS_OUTPUT.PUT_LINE('SI Final Amount: Rs.' ||
        ROUND(v_si_amount, 2));
    DBMS_OUTPUT.PUT_LINE('CI Final Amount: Rs.' ||
        ROUND(v_ci_amount, 2));

    IF v_diff_pct > 10 THEN
        DBMS_OUTPUT.PUT_LINE(
            'Tip: FD is better than savings account for this amount.'
        );
    END IF;
END;
