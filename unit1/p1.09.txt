Problem Statement: Build a simple electricity bill for Torrent Power. Declare: v_units NUMBER, c_rate CONSTANT NUMBER := 5 (Rs. per unit), c_fixed CONSTANT NUMBER := 70 (fixed monthly charge). Calculate: energy_charge = units * rate, total = energy_charge + fixed charge. Print a mini bill: consumer name (your name), units consumed, energy charge, fixed charge, and total, each on its own line using DBMS_OUTPUT.PUT_LINE.

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_name VARCHAR2(50) := 'Harvi Upadhyay';
    v_units NUMBER := 610;
    c_rate CONSTANT NUMBER := 5;
    c_fixed CONSTANT NUMBER := 70;
    v_energy NUMBER;
    v_total NUMBER;
BEGIN
    v_energy := v_units * c_rate;
    v_total := v_energy + c_fixed;

    DBMS_OUTPUT.PUT_LINE('----- TORRENT POWER BILL -----');
    DBMS_OUTPUT.PUT_LINE('Consumer: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('Units: ' || v_units);
    DBMS_OUTPUT.PUT_LINE('Energy Charge: Rs.' || v_energy);
    DBMS_OUTPUT.PUT_LINE('Fixed Charge: Rs.' || c_fixed);
    DBMS_OUTPUT.PUT_LINE('Total: Rs.' || v_total);
END;
