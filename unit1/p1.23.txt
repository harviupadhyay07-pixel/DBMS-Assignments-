DECLARE
    -- Bug 1 fixed:
    -- := is used for assignment instead of =.
    v_name VARCHAR2(50) := 'Harvi Upadhyay';

    -- Bug 2 fixed:
    -- VARCHAR2(50) is large enough for the full name.

    -- Bug 3 fixed:
    -- CONSTANT must be initialized when declared.
    c_bonus_rate CONSTANT NUMBER := 0.1;

    -- Bug 4 fixed:
    -- Semicolon added after the declaration.
    v_sal NUMBER := 76000;

    v_bonus NUMBER;
    v_fname employees.first_name%TYPE;
BEGIN
    v_bonus := v_sal * c_bonus_rate;

    -- Bug 5 fixed:

    SELECT first_name
    INTO v_fname
    FROM employees
    WHERE employee_id = 100;

    DBMS_OUTPUT.PUT_LINE('Name: ' || v_name);
    DBMS_OUTPUT.PUT_LINE(
        'Employee: ' || v_fname
    );
    DBMS_OUTPUT.PUT_LINE(
